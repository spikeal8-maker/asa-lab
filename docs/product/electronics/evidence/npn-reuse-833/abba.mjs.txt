import { writeFile } from 'node:fs/promises';
import { createHash } from 'node:crypto';
const modules={ A:await import('./before.mjs'), B:await import('./after.mjs') };
const encode=v=>JSON.stringify(v,(_,x)=>x instanceof Map?{map:[...x]}:x instanceof Set?{set:[...x]}:x);
const run=variant=>{
 globalThis.__npnProbe={calls:0,hits:0}; const start=performance.now();
 const result=modules[variant].run();const elapsedMs=performance.now()-start;
 const raw=encode(result);return{variant,elapsedMs,...globalThis.__npnProbe,bytes:Buffer.byteLength(raw),sha256:createHash('sha256').update(raw).digest('hex')};
};
const warmup=[run('A'),run('B')];const samples=['A','B','B','A'].map(run);
const receipt={warmup,samples};
await writeFile(new URL('./abba.json',import.meta.url),JSON.stringify(receipt,null,2));console.log(JSON.stringify(receipt));
