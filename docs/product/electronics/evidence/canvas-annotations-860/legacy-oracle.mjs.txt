import { createRequire } from 'node:module';
import { mkdirSync, writeFileSync } from 'node:fs';
import { gzipSync } from 'node:zlib';
import { createHash } from 'node:crypto';
import { pathToFileURL } from 'node:url';
const require = createRequire(import.meta.url);
const { build } = createRequire(require.resolve('vite'))('esbuild');
const dir = 'test-results/canvas-annotations-860';
mkdirSync(dir, { recursive: true });
const name = process.argv[2] ?? 'before';
await build({ entryPoints: ['contexts/electronics/index.ts'], bundle: true, platform: 'node', format: 'esm', outfile: `${dir}/${name}-runtime.mjs`, logLevel: 'silent' });
const m = await import(pathToFileURL(`${process.cwd()}/${dir}/${name}-runtime.mjs`).href);
const inputs = [1, 2, 3, 4].flatMap(schemaVersion => [0, 1, 2].map(n => ({ schemaVersion, components: [], connections: [], viewport: { x: n * 10, y: -n, zoom: 1 + n / 10 }, simulation: { running: !!n, maxIterations: 24 + n } })));
inputs.push({ schemaVersion: 4, components: [{ id: 'source', kind: 'source', position: { x: 0, y: 0 }, value: 5 }, { id: 'resistor', kind: 'resistor', position: { x: 100, y: 0 }, value: 1000 }], connections: [{ id: 'p', from: { componentId: 'source', terminal: 'a' }, to: { componentId: 'resistor', terminal: 'a' } }, { id: 'n', from: { componentId: 'resistor', terminal: 'b' }, to: { componentId: 'source', terminal: 'b' } }], viewport: { x: 0, y: 0, zoom: 1 }, simulation: { running: false, maxIterations: 24 } });
const data = inputs.map(input => { const parsed = m.parseElectronicsDocument(input); return { input, parsed, ...(parsed.ok ? { digest: m.simulationInputDigest(parsed.document), result: m.analyseCircuit(parsed.document) } : {}) }; });
const raw = Buffer.from(JSON.stringify(data));
const receipt = { cases: data.length, bytes: raw.length, sha256: createHash('sha256').update(raw).digest('hex'), baseline: 'ee867abe3a2e61e0fbcf977eb9f3f317d31bdf83' };
writeFileSync(`${dir}/${name}.json.gz`, gzipSync(raw, { mtime: 0 }));
writeFileSync(`${dir}/${name}-receipt.json`, JSON.stringify(receipt, null, 2) + '\n');
console.log(receipt);
