import { readFile, writeFile, mkdir } from 'node:fs/promises';
import { resolve, dirname } from 'node:path';
import { createRequire } from 'node:module';
import { pathToFileURL } from 'node:url';
import { gzipSync } from 'node:zlib';
import { createHash } from 'node:crypto';
import process from 'node:process';
import {execFileSync} from 'node:child_process';
const require = createRequire(import.meta.url);
const { build } = createRequire(require.resolve('vite'))('esbuild');
const out = resolve('test-results/void-functions-824');
await mkdir(out, { recursive: true });
const file = resolve('contexts/electronics/domain/arduino-program-runtime.ts');
const baseline=process.argv[2]==='baseline';
const runtimeSource = baseline ? execFileSync('git',['show','00301affb4a66e2e19ee706cdb25b91294b987be:contexts/electronics/domain/arduino-program-runtime.ts'],{encoding:'utf8'}) : await readFile(file,'utf8');
await build({ stdin: { contents: runtimeSource + '\nexport {compileArduinoProgram as oracleCompile};', resolveDir: dirname(file), loader: 'ts' }, bundle: true, platform: 'node', format: 'esm', outfile: resolve(out, 'oracle-runtime.mjs') });
const runtime = await import(pathToFileURL(resolve(out, 'oracle-runtime.mjs')).href + '?t=' + Date.now());
const ultra = `float readUltrasonicCm(int triggerPin, int echoPin) {
  pinMode(triggerPin, OUTPUT);
  digitalWrite(triggerPin, LOW);
  delayMicroseconds(2);
  digitalWrite(triggerPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(triggerPin, LOW);
  pinMode(echoPin, INPUT);
  return pulseIn(echoPin, HIGH) * 0.01723;
}`;
const cases = [
'int v=0;void loop(){v=min(8,3);delay(1);}',
'int v=0;void loop(){v=max(-8,3);delay(1);}',
'int v=0;void loop(){v=abs(-13);delay(1);}',
'int v=0;void loop(){v=constrain(23,0,10);delay(1);}',
'int n=1;int v=0;void loop(){v=min(n++,5)+max(n++,2)+abs(-n)+constrain(n++,1,4);delay(1);}',
];
const reader = s => ({ a0: 2.5, d3: s.simulationTimeMs >= 0.006 && s.simulationTimeMs < 0.009 ? 5 : 0, d8: 0 });
const results = [];
for (const source of cases) {
  const entry = { source, compilation: JSON.parse(JSON.stringify(runtime.oracleCompile(source))), runs: [] };
  for (const budget of [1, 2, 7]) {
    let state;
    const ticks = [];
    for (const horizon of [0, 0.02, 0.12]) {
      const advances = [];
      for (let step = 0; step < 1000; step++) {
        const next = runtime.advanceClockedArduinoRuntime(source, {}, horizon, state, reader, { instructionBudget: budget, pulseInputSample: true });
        advances.push(JSON.parse(JSON.stringify(next)));
        state = JSON.parse(JSON.stringify(next.state));
        if (next.executionStatus !== 'yielded') break;
        if (step === 999) throw new Error('Oracle bounded step limit');
      }
      ticks.push({ horizon, advances });
    }
    entry.runs.push({ profile: 'instruction-us-v1', budget, ticks });
  }
  let legacy;
  entry.runs.push({ profile: 'legacy-ms-v1', ticks: [0, 1, 2].map(horizon => {
    const next = runtime.advanceArduinoRuntime(source, {}, horizon, legacy, reader);
    legacy = JSON.parse(JSON.stringify(next.state));
    return { horizon, advance: JSON.parse(JSON.stringify(next)) };
  }) });
  entry.compilationAfter = JSON.parse(JSON.stringify(runtime.oracleCompile(source)));
  results.push(entry);
}
const bytes = Buffer.from(JSON.stringify(results));
const label = baseline ? 'numeric-macros-baseline' : 'numeric-macros-after';
await writeFile(resolve(out, label+'.json.gz'), gzipSync(bytes, { mtime: 0 }));
const receipt = { cases: results.length, bytes: bytes.length, sha256: createHash('sha256').update(bytes).digest('hex'), runtimeSha256: createHash('sha256').update(runtimeSource).digest('hex') };
await writeFile(resolve(out, label+'-receipt.json'), JSON.stringify(receipt,null,2)+'\n');
console.log(JSON.stringify(receipt));
